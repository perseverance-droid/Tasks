# 成员管理系统源码

```go
package main

import (
	"net/http"
	"strconv"

	"github.com/gin-gonic/gin"
	"golang.org/x/crypto/bcrypt"
	"gorm.io/driver/mysql"
	"gorm.io/gorm"

)

var db *gorm.DB

type Member struct {
	gorm.Model
	Username string `json:"username" gorm:"unique"`
	Password string `json:"-"`
	Name     string `json:"name"`
	Phone    string `json:"phone"`
	Email    string `json:"email"`
	Status   int    `json:"status"` // 0待审核 1正常
	Role     string `json:"role"`   // admin/user
}

func initDB() {
	dsn := "root:123456@tcp(127.0.0.1:3306)/member_db?charset=utf8mb4&parseTime=True&loc=Local"
	var err error
	db, err = gorm.Open(mysql.Open(dsn), &gorm.Config{})
	if err != nil {
		panic("数据库连接失败")
	}
	db.AutoMigrate(&Member{})
}

func (m *Member) Encrypt() {
	hash, _ := bcrypt.GenerateFromPassword([]byte(m.Password), bcrypt.DefaultCost)
	m.Password = string(hash)
}

func (m *Member) Check(pwd string) bool {
	return bcrypt.CompareHashAndPassword([]byte(m.Password), []byte(pwd)) == nil
}

func Register(c *gin.Context) {
	var m Member
	c.ShouldBindJSON(&m)
	m.Encrypt()
	m.Status = 0
	m.Role = "user"
	db.Create(&m)
	c.JSON(200, gin.H{"msg": "注册成功"})
}

func Login(c *gin.Context) {
	var req struct {
		Username string `json:"username"`
		Password string `json:"password"`
	}
	c.ShouldBindJSON(&req)

	var m Member
	db.Where("username=?", req.Username).First(&m)
	if !m.Check(req.Password) {
		c.JSON(401, gin.H{"msg": "密码错误"})
		return
	}
	c.JSON(200, gin.H{"msg": "登录成功", "role": m.Role})

}

func List(c *gin.Context) {
	var list []Member
	db.Find(&list)
	c.JSON(200, list)
}

func Audit(c *gin.Context) {
	id, _ := strconv.Atoi(c.Param("id"))
	db.Model(&Member{}).Where("id=?", id).Update("status", 1)
	c.JSON(200, gin.H{"msg": "审核成功"})
}

func Delete(c *gin.Context) {
	id, _ := strconv.Atoi(c.Param("id"))
	db.Delete(&Member{}, id)
	c.JSON(200, gin.H{"msg": "删除成功"})
}

func UpdateInfo(c *gin.Context) {
	var req struct {
		Name  string `json:"name"`
		Phone string `json:"phone"`
		Email string `json:"email"`
	}
	c.ShouldBindJSON(&req)
	username := c.GetHeader("username")
	db.Model(&Member{}).Where("username=?", username).Updates(&req)
	c.JSON(200, gin.H{"msg": "修改成功"})
}

func main() {
	initDB()
	r := gin.Default()

	r.POST("/register", Register)
	r.POST("/login", Login)
	
	admin := r.Group("/admin")
	{
		admin.GET("/members", List)
		admin.PUT("/audit/:id", Audit)
		admin.DELETE("/member/:id", Delete)
	}
	
	user := r.Group("/user")
	{
		user.PUT("/info", UpdateInfo)
	}
	
	r.Run(":8080")

}
```
